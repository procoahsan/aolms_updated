import * as XLSX from 'xlsx';
import { ILogisticsRecord } from './logistics-excel-parser.util';

export interface IComparedRecord extends ILogisticsRecord {
  // Team sheet fields
  teamSheetFound: boolean;       // was the ticket matched in team sheet?
  teamOnt: string;               // Replaced ONT
  teamOldSn: string;             // OLD ONT Serial No
  teamNewSn: string;             // New ONT Serial No
  teamOntBox: string;            // Fiber Termination Box (FTB)
  teamName: string;              // Team Name
  // Comparison result
  isMatch: boolean;              // true if ticket found in team sheet
  remarks: string[];             // list of mismatch descriptions
}

export interface ITeamComparisonResult {
  data: IComparedRecord[];
  availableDates: { label: string; value: string }[];
  selectedDate: string;
}

/**
 * Excel serial number (days since 1900-01-00) -> JS Date at UTC midnight.
 */
function serialToUTCDate(serial: number): Date {
  return new Date(Math.round((serial - 25569) * 86400 * 1000));
}

/**
 * Format an Excel serial date to a readable date string (e.g. "3-Apr").
 */
function serialToDateLabel(serial: number): string {
  const date = serialToUTCDate(serial);
  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  return `${date.getUTCDate()}-${months[date.getUTCMonth()]}`;
}

/**
 * Extract unique dates from team file "Form Responses 1" sheet timestamps.
 * Returns sorted (descending) array of { label, serialFloor }.
 */
function extractTeamDates(teamBuffer: Buffer): { label: string; serialFloor: number }[] {
  const workbook = XLSX.read(teamBuffer, { type: 'buffer' });
  
  // Find "Form Responses 1" sheet
  const sheetName = workbook.SheetNames.find(
    (n) => n.toLowerCase().includes('form responses'),
  ) || workbook.SheetNames[0];
  
  const sheet = workbook.Sheets[sheetName];
  const rows = XLSX.utils.sheet_to_json<any>(sheet, { defval: '' });

  const dateSet = new Map<number, string>(); // serialFloor -> label

  for (const row of rows) {
    const ts = row['Timestamp'];
    if (typeof ts === 'number' && ts > 0) {
      const floor = Math.floor(ts);
      if (!dateSet.has(floor)) {
        dateSet.set(floor, serialToDateLabel(ts));
      }
    }
  }

  // Sort descending by serial
  const sorted = Array.from(dateSet.entries())
    .sort((a, b) => b[0] - a[0])
    .map(([serialFloor, label]) => ({ label, serialFloor }));

  return sorted;
}

/**
 * Compare logistics records with team sheet.
 * Only checks if ticket is found in team sheet — no field-level comparison.
 * Filters team entries by selectedDate (from team timestamps).
 */
export function compareLogisticsWithTeam(
  logisticsRecords: ILogisticsRecord[],
  logisticsRawBuffer: Buffer,
  teamBuffer: Buffer,
  selectedDateSerial?: number,
): ITeamComparisonResult {
  // --- Extract available dates from team file ---
  const teamDates = extractTeamDates(teamBuffer);

  // Determine selected date (default: 2nd latest from team timestamps)
  let selectedSerial = selectedDateSerial;
  if (selectedSerial === undefined || selectedSerial === null) {
    if (teamDates.length >= 2) {
      selectedSerial = teamDates[1].serialFloor; // 2nd latest
    } else if (teamDates.length === 1) {
      selectedSerial = teamDates[0].serialFloor;
    } else {
      selectedSerial = 0;
    }
  }

  // --- Parse team sheet ---
  const teamWorkbook = XLSX.read(teamBuffer, { type: 'buffer' });
  const teamSheetName = teamWorkbook.SheetNames.find(
    (n) => n.toLowerCase().includes('form responses'),
  ) || teamWorkbook.SheetNames[0];
  const teamSheet = teamWorkbook.Sheets[teamSheetName];
  const teamRaw = XLSX.utils.sheet_to_json<any>(teamSheet, { defval: '' });

  // Build set of ticket numbers from team entries matching the selected date
  const teamTickets = new Set<string>();
  const teamNameMap = new Map<string, string>(); // ticketNo -> teamName

  for (const row of teamRaw) {
    const tNo = String(row['SR / Ticket number'] ?? '').trim();
    const ts = row['Timestamp'];
    const dateSerial = typeof ts === 'number' ? Math.floor(ts) : 0;

    if (tNo && dateSerial === selectedSerial) {
      teamTickets.add(tNo);
      const teamName = String(row['Team Name'] ?? '').trim();
      if (teamName) teamNameMap.set(tNo, teamName);
    }
  }

  // --- Compare: only check if ticket exists in team sheet ---
  const comparedData: IComparedRecord[] = logisticsRecords.map((logRec) => {
    const tNo = logRec.ticketNo;
    const found = teamTickets.has(tNo);

    return {
      ...logRec,
      teamSheetFound: found,
      teamOnt: '',
      teamOldSn: '',
      teamNewSn: '',
      teamOntBox: '',
      teamName: found ? (teamNameMap.get(tNo) || '') : '',
      isMatch: found,
      remarks: found ? [] : ['Ticket not found in Team Sheet'],
    };
  });

  // Build available dates for frontend
  const availableDates = teamDates.map(d => ({
    label: d.label,
    value: String(d.serialFloor),
  }));

  return {
    data: comparedData,
    availableDates,
    selectedDate: String(selectedSerial),
  };
}
